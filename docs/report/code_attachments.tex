\section{Time-respecting Neighbour Sampling}
From \texttt{src/argus/graph/sampling.py}: a scored transfer only sees transfers at or before its own timestamp.

\begin{lstlisting}[language=Python]
def make_loader(graph, edge_ids, gc: dict, shuffle: bool, batch_size: int) -> LinkNeighborLoader:
    """Loader yielding a sampled subgraph around each batch of transfers `edge_ids`.

    With gc["temporal"] (default on), a scored transfer only sees transfers with
    time <= its own, so there is no look-ahead even on the full test-period graph.
    PyG makes temporal sampling disjoint: one subgraph per scored transfer.
    """
    nn_cfg = {et: gc["num_neighbors"][et[1]] for et in graph.edge_types}
    temporal = {}
    if gc.get("temporal", True):
        temporal = dict(time_attr="time", edge_label_time=graph[TRANSFER].time[edge_ids],
                        temporal_strategy=gc.get("temporal_strategy", "last"))
    return LinkNeighborLoader(
        graph,
        num_neighbors=nn_cfg,
        edge_label_index=(TRANSFER, graph[TRANSFER].edge_index[:, edge_ids]),
        edge_label=edge_ids,            # carry global edge ids through the sampler
        batch_size=batch_size,
        shuffle=shuffle,
        num_workers=gc.get("num_workers", 0),
        **temporal,
    )
\end{lstlisting}

\section{An Attack Operator: Structuring}
From \texttt{src/argus/twin/attacks.py}: each laundering transfer is split into 2--4 smaller transfers minutes apart.

\begin{lstlisting}[language=Python]
    def split(self, rt, acc):
        rows = []
        for r in rt.itertuples(index=False):
            d = r._asdict()
            if r.role != "ring":
                rows.append(d); continue
            k = int(self.rng.integers(2, 5))
            for i, part in enumerate(self.rng.dirichlet(np.ones(k) * 3) * r.amount_paid):
                rows.append({**d, "ts": r.ts + pd.Timedelta(minutes=3 * i + int(self.rng.integers(0, 3))),
                             "amount_paid": round(part, 2), "amount_received": round(part, 2)})
        return pd.DataFrame(rows), acc
\end{lstlisting}

\section{Adaptive Attack and Retrain Protocol}
From \texttt{src/argus/twin/protocol.py}: each round the attacker adds the operator that lowers validation recall most; degradation $D$ and recovery $R$ are recorded (logging lines omitted).

\begin{lstlisting}[language=Python]
    out["adaptive"] = []
    det, playbook, before = det0, [], base["f1"]
    for t in range(1, rounds + 1):
        unused = [op for op in OPERATORS if op not in playbook]
        if not unused:
            break
        probe = {op: metrics(det, exp.version(playbook, playbook + [op], playbook), rows="val")["recall"]
                 for op in unused}
        chosen = min(probe, key=probe.get)
        new = playbook + [chosen]
        att = metrics(det, exp.version(playbook, playbook, new))
        v = exp.version(new, new, new)
        det = make_detector().fit(*v)
        ret = metrics(det, v)
        rec = {"round": t, "added": chosen, "playbook": new, "probe_val_recall": probe,
               "before_f1": round(before, 4), "attack": _short(att), "retrained": _short(ret),
               "D": round(before - att["f1"], 4), "R": round(ret["f1"] - att["f1"], 4)}
        out["adaptive"].append(rec)
        playbook, before = new, ret["f1"]
\end{lstlisting}

\section{Grounding Check for Stated Counts}
From \texttt{src/argus/agent/report.py}: counts written in the language-model narrative must equal the case indicator they describe.

\begin{lstlisting}[language=Python]
def count_claims_check(narrative: str, case: dict) -> list:
    ind = case["indicators"]
    n_exit, n_entry = len(case["exit_accounts"]), len(case["entry_accounts"])
    allowed = {
        "dormant": {ind["dormant_accounts"]},
        "bank": {ind["banks"]},
        "account": {ind["accounts"], n_exit, n_entry, ind["accounts"] - n_entry, ind["accounts"] - n_entry - n_exit,
                    ind["dormant_accounts"], ind["max_fan_out"], ind["max_fan_in"]},
        "transfer": {ind["transfers"], *ind["payment_formats"].values(), ind["max_fan_out"], ind["max_fan_in"]},
    }
    bad = []
    for m in _COUNT.finditer(narrative):
        raw, qual, noun = m.group(1).lower(), (m.group(2) or "").lower(), m.group(3).lower()
        n = _WORDS[raw] if raw in _WORDS else int(raw)
        key = ("dormant" if noun == "dormant" else "bank" if noun.startswith("bank")
               else "account" if noun.startswith("account") else "transfer")
        if key == "account" and "dormant" in m.group(0).lower():
            key = "dormant"
        if n not in allowed[key]:
            bad.append(m.group(0))
    return bad
\end{lstlisting}
